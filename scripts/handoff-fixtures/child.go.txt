// This source is copied to a temporary module by Test-LauncherHandoff.ps1.
// It records only allowlisted fixture values; it never dumps the process environment.
package main

import (
	"encoding/json"
	"fmt"
	"io"
	"os"
	"runtime"
	"strconv"
)

type environmentValue struct {
	Value   string `json:"value"`
	Present bool   `json:"present"`
}

type observation struct {
	SchemaVersion int                         `json:"schemaVersion"`
	Executable    string                      `json:"executable"`
	Arguments     []string                    `json:"arguments"`
	WorkingDir    string                      `json:"workingDirectory"`
	Environment   map[string]environmentValue `json:"environment"`
	Stdin         string                      `json:"stdin"`
	GOOS          string                      `json:"goos"`
	GOARCH        string                      `json:"goarch"`
}

func observeEnvironment(name string) environmentValue {
	value, present := os.LookupEnv(name)
	return environmentValue{Value: value, Present: present}
}

func main() {
	stdin, err := io.ReadAll(os.Stdin)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
	workingDir, err := os.Getwd()
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
	executable, err := os.Executable()
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
	output := os.Getenv("HS_HANDOFF_OUTPUT")
	if output == "" {
		fmt.Fprintln(os.Stderr, "HS_HANDOFF_OUTPUT is required")
		os.Exit(2)
	}

	observed := observation{
		SchemaVersion: 3,
		Executable:    executable,
		Arguments:     append([]string(nil), os.Args[1:]...),
		WorkingDir:    workingDir,
		Environment: map[string]environmentValue{
			"HS_HANDOFF_SELECTED":  observeEnvironment("HS_HANDOFF_SELECTED"),
			"HS_HANDOFF_EMPTY":     observeEnvironment("HS_HANDOFF_EMPTY"),
			"HS_HANDOFF_EXIT_CODE": observeEnvironment("HS_HANDOFF_EXIT_CODE"),
			"HS_HANDOFF_INHERITED": observeEnvironment("HS_HANDOFF_INHERITED"),
		},
		Stdin:  string(stdin),
		GOOS:   runtime.GOOS,
		GOARCH: runtime.GOARCH,
	}
	if runtime.GOOS == "windows" {
		observed.Environment["OS"] = observeEnvironment("OS")
	}
	encoded, err := json.Marshal(observed)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
	if err := os.WriteFile(output, append(encoded, '\n'), 0o600); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(2)
	}
	fmt.Fprintln(os.Stdout, "HS_HANDOFF_STDOUT_SENTINEL")
	fmt.Fprintln(os.Stderr, "HS_HANDOFF_STDERR_SENTINEL")

	exitCode, err := strconv.Atoi(os.Getenv("HS_HANDOFF_EXIT_CODE"))
	if err != nil {
		fmt.Fprintln(os.Stderr, "HS_HANDOFF_EXIT_CODE must be an integer")
		os.Exit(2)
	}
	os.Exit(exitCode)
}
